20: def load_card_definitions(path: Path | None = None) -> dict[str, CardDefinition]:
21:     with (path or DATA_DIR / "anime_bedriegers_bedrogen.json").open(encoding="utf-8") as source:
22:         raw_data = json.load(source)
23:     raw_cards = raw_data.get("cards") if isinstance(raw_data, dict) else None
24:     if not isinstance(raw_cards, list):
25:         raise ValueError("Character data must contain a 'cards' list")
26: 
27:     definitions: dict[str, CardDefinition] = {}
28:     duplicate_ids: set[str] = set()
29:     missing_ids = 0
30:     for index, raw in enumerate(raw_cards):
31:         raw_id = raw.get("id") if isinstance(raw, dict) else None
32:         if raw_id is None or not str(raw_id).strip():
33:             missing_ids += 1
34:             logger.warning("Character roster entry %d is missing an ID; entry skipped.", index + 1)
35:             continue
36:         card_id = str(raw_id)
37:         if card_id in definitions:
38:             duplicate_ids.add(card_id)
39:             continue
40:         ability_raw = raw.get("ability")
41:         definition = CardDefinition(
42:             id=card_id,
43:             name=str(raw["name"]),
44:             image=str(raw.get("image") or f"assets/art/characters/{card_id}.png"),
45:             base_points=int(raw["base_points"]),
46:             effect_name=str(raw.get("effect_name", "")),
47:             effect_text=str(raw.get("effect_text", "")),
48:             full_card_art=str(raw.get("full_card_art", "")),
49:             # Roster classification drives elimination allowed_types filters.
50:             type=str(raw.get("type", "")),
51:             # Malformed/unsupported abilities fail loudly with the card named.
52:             ability=(
53:                 parse_ability(ability_raw, context=f"Character {card_id}")
54:                 if ability_raw is not None
55:                 else None
56:             ),
57:         )
58:         if not definition.name or not definition.image:
59:             raise ValueError(f"Character {definition.id} needs a name and image path")
60:         definitions[definition.id] = definition
61:     if duplicate_ids:
62:         logger.warning(
63:             "Character roster has duplicate IDs (duplicate entries skipped): %s",
64:             ", ".join(sorted(duplicate_ids)),
65:         )
66:     if missing_ids:
67:         logger.warning("Character roster has %d entry/entries missing an ID.", missing_ids)
68:     if len(definitions) < EXPECTED_ROSTER_SIZE:
69:         logger.warning(
70:             "Character roster is missing %d unique character(s): found %d, expected %d.",
71:             EXPECTED_ROSTER_SIZE - len(definitions), len(definitions), EXPECTED_ROSTER_SIZE,
72:         )
73:     elif len(definitions) > EXPECTED_ROSTER_SIZE:
74:         logger.warning(
75:             "Character roster has %d extra unique character(s): found %d, expected %d.",
76:             len(definitions) - EXPECTED_ROSTER_SIZE, len(definitions), EXPECTED_ROSTER_SIZE,
77:         )
78:     if not definitions:
79:         raise ValueError("At least one card definition is required")
80:     # Character references and duplicate ability ids can only be checked once
81:     # the complete roster is known.
82:     validate_ability_references(definitions)
83:     return definitions
