33| def scripted_engine(capacity=3, seed=4):
34|     """Engine with scrubbed pools so only scripted cards reach the board."""
35|     engine = GameEngine(
36|         GameConfig(
37|             deck_size=40, initial_hand_size=5, draw_per_turn=1,
38|             capacity_min=capacity, capacity_max=capacity, rounds_per_match=2,
39|         ),
40|         seed=seed,
41|     )
42|     serial = 0
43| 
44|     def _scrub(player):
45|         nonlocal serial
46|         for index in range(len(player.draw_pile)):
47|             serial += 1
48|             definition_id = FILLERS[index % len(FILLERS)]
49|             player.draw_pile[index] = CardInstance(
50|                 f"filler-{serial}", definition_id, player.id,
51|                 current_points=engine.cards[definition_id].base_points,
52|             )
53| 
54|     for player in engine.state.players:
55|         _scrub(player)
56|     return engine

59| def give(player, engine, *definition_ids):
60|     player.hand = []
61|     for definition_id in definition_ids:
62|         definition = engine.cards[definition_id]
63|         player.hand.append(CardInstance(
64|             f"{player.id}-{definition_id}-{len(player.hand)}",
65|             definition_id, player.id, current_points=definition.base_points,
66|         ))

69| def place(engine, card, row_index, slot_index):
70|     active = engine.state.active_player
71|     assert active.id == card.owner_id, "test sequence must follow turn order"
72|     result = engine.play_card(active.id, card.id, engine.state.rows[row_index].id, slot_index)
73|     assert result.accepted, result.reason
74|     return result

77| def card(engine, definition_id, owner_id):
78|     definition = engine.cards[definition_id]
79|     return CardInstance(
80|         f"unit-{definition_id}-{owner_id}", definition_id, owner_id,
81|         current_points=definition.base_points,
82|     )

85| def reveal_all_but(engine):
86|     """Reveal every placed card so only scripted hidden states remain."""
87|     for row in engine.state.rows:
88|         for slot in row.slots:
89|             if slot is not None and not slot.revealed:
90|                 slot.revealed = True

93| def targets_of(manager, state, attacker, targeting):
94|     return [
95|         (row.id, index, placed.definition_id)
96|         for row, index, placed in manager.get_valid_targets(state, attacker, targeting)
97|     ]

102| def _board_with_two_sides():
103|     """Capacity-2 board: p1 holds row0/s0 + row1/s0, p2 holds row0/s1 + row1/s1."""
104|     engine = scripted_engine(capacity=2)
105|     p1, p2 = engine.state.players
106|     a = card(engine, KAKASHI, p1.id)    # 7 / Fighter, the attacker
107|     p1_filler = card(engine, ITACHI, p1.id)
108|     b = card(engine, GO, p2.id)         # 10 / GOAT
109|     c = card(engine, HISOKA, p2.id)     # 8 / Fighter
110|     p1.hand = [a, p1_filler]
111|     p2.hand = [b, c]
112|     place(engine, a, 0, 0)
113|     place(engine, b, 0, 1)          # reveals a
114|     place(engine, p1_filler, 1, 0)
115|     place(engine, c, 1, 1)          # reveals p1_filler
116|     reveal_all_but(engine)
117|     return engine, a, b, c

166| def test_own_side_hidden_and_eliminated_cards_are_never_eligible():
167|     engine, attacker, _, _ = _board_with_two_sides()
168|     manager = EliminationManager(engine.cards)
169|     state = engine.state
170|     p2 = state.players[1]
171|     # Same-owner cards are excluded even when revealed and strong.
172|     assert all(
173|         placed.owner_id == p2.id
174|         for _, _, placed in manager.get_valid_targets(state, attacker, EliminationTargeting())
175|     )
176|     # Hidden opposing cards are excluded.
177|     hidden = card(engine, LUFFY, p2.id)
178|     state.rows[1].slots[1] = hidden
179|     assert hidden.id not in {
180|         placed.id for _, _, placed in manager.get_valid_targets(
181|             state, attacker, EliminationTargeting(),
182|         )
183|     }
184|     # Already-eliminated cards drop out of the target list.
185|     b = next(
186|         slot for slot in state.rows[0].slots
187|         if slot is not None and slot.definition_id == GO
188|     )
189|     manager.eliminate_target(state, b, attacker)
190|     assert b.id not in {

199| def fill_rest(engine):
200|     """Place drawn fillers until placement ends; settle any pauses."""
201|     while engine.state.phase == GamePhase.PLACEMENT:
202|         resolve_eliminations(engine)
203|         if engine.state.phase != GamePhase.PLACEMENT:
204|             break
205|         active = engine.state.active_player
206|         row_id, slot_index = engine.legal_slots()[0]
207|         result = engine.play_card(active.id, active.hand[0].id, row_id, slot_index)
208|         assert result.accepted, result.reason
209|     resolve_eliminations(engine)

224| def test_final_reveal_pause_resumes_after_selecting_a_target():
225|     engine = scripted_engine(capacity=3)
226|     p1, p2 = engine.state.players
227|     give(p1, engine, GO, LUFFY, KILLUA)
228|     give(p2, engine, KAKASHI, HISOKA, ITACHI)
229|     go, luffy, killua = p1.hand
230|     kakashi, hisoka, itachi = p2.hand
231|     place(engine, go, 0, 0)              # GO (p1, hidden)
232|     place(engine, kakashi, 0, 1)         # KAKASHI reveals GO
233|     place(engine, killua, 0, 2)          # KILLUA reveals KAKASHI; row0 full
234|     place(engine, hisoka, 1, 0)
235|     place(engine, luffy, 1, 1)           # reveals HISOKA
236|     place(engine, itachi, 1, 2)          # reveals LUFFY
237|     fill_rest(engine)
238|     assert engine.state.phase == GamePhase.REVEAL
239| 
240|     reveal = engine.reveal_next_card()       # first hidden: row0/s2 = KILLUA
241|     assert reveal.accepted
242|     pending = engine.state.pending_elimination
243|     assert pending is not None
244|     assert pending["attacker_card_id"] == killua.id
245|     assert pending["attacker_owner_id"] == "player-1"
246|     # Row-major order: KAKASHI (row0/s1) first, then HISOKA (row1/s0).
247|     assert [(t["row_id"], t["slot_index"]) for t in pending["valid_targets"]] == [
248|         (engine.state.rows[0].id, 1),
249|         (engine.state.rows[1].id, 0),
250|     ]
251|     # The reveal flow is paused: another reveal is rejected until resolved.
252|     paused = engine.reveal_next_card()
253|     assert not paused.accepted
254|     assert "elimination" in paused.reason
255| 
256|     target = pending["valid_targets"][0]
257|     result = engine.select_elimination_target(target["row_id"], target["slot_index"])
258|     assert result.accepted, result.reason
259|     assert engine.state.pending_elimination is None
260|     eliminated = engine.state.rows[0].slots[1]
261|     assert eliminated.definition_id == KAKASHI
262|     assert eliminated.eliminated is True
263|     assert eliminated.current_points == 0
264|     assert engine.state.rows[0].eliminated_slots == [1]
265|     elimination_events = [event for event in result.events if event.kind == "elimination"]
266|     assert len(elimination_events) == 1
267|     assert elimination_events[0].data["attacker_name"] == "Killua Zoldyck"
268|     assert elimination_events[0].data["target_name"] == "Kakashi Hatake"
269| 
270|     # Reveal flow resumes and finishes the round normally.
271|     while engine.state.phase == GamePhase.REVEAL:
272|         assert engine.reveal_next_card().accepted
273|     assert engine.state.phase == GamePhase.ROW_SCORING
