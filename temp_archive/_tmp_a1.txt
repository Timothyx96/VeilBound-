2: """Central character-ability system: JSON definitions and reveal-time resolution.
3: 
4: Abilities are declared per character in the roster JSON and executed only by
5: :class:`AbilityManager`, which the authoritative
6: :class:`~veilbound.shared.rules.GameEngine` owns. Rendering, animation, and
7: scoring code never implement ability rules.
8: 
9: Resolution order when a card becomes revealed (documented contract):
10: 
11: 1. The board state is updated first (``card.revealed = True``) by the caller.
12: 2. **Phase A** - previously registered abilities (owner revealed earlier, not
13:    yet eligible) are re-evaluated in registration order - which equals owner
14:    reveal order - ties broken by ascending ability id.
15: 3. **Phase B** - the newly revealed card's own on-reveal abilities run next.
16: 4. **Phase C** - each eligible ability applies its effects exactly once
17:    (guarded by the owner card instance's ``activated_abilities`` record) and
18:    its presentation is emitted immediately after that ability's effects.
19: 
20: Duplicate activation is prevented by both the pending-registration bookkeeping
21: and the per-card-instance activation record, so repeated condition checks,
22: animation frames, or duplicated events cannot grant the bonus twice. A
23: re-entrancy guard makes circular ability chains impossible: a nested
24: resolution pass is logged and skipped.
25: """

143:     value = raw.get(key)
144:     if not isinstance(value, str) or not value.strip():
145:         raise AbilityValidationError(f"{context}: '{key}' must be a non-empty string.")
146:     return value.strip()

149: def parse_ability(raw: Any, *, context: str) -> AbilityDefinition:
150:     """Parse and validate one ``ability`` object from character JSON.
151: 
152:     Raises :class:`AbilityValidationError` naming the offending character and
153:     field so bad data is caught while character data loads.
154:     """
155:     if not isinstance(raw, dict):
156:         raise AbilityValidationError(f"{context}: 'ability' must be a JSON object.")
157:     ability_id = _require_str(raw, "id", context)
158:     name = _require_str(raw, "name", context)
159:     description = str(raw.get("description", ""))
160: 
161:     trigger = _require_str(raw, "trigger", context)
162:     if trigger not in SUPPORTED_TRIGGERS:
163:         raise AbilityValidationError(
164:             f"{context}: unsupported trigger {trigger!r} for ability {ability_id!r}; "
165:             f"supported triggers: {sorted(SUPPORTED_TRIGGERS)}."
166:         )
167: 
168:     conditions_raw = raw.get("conditions", [])
169:     if not isinstance(conditions_raw, list):
170:         raise AbilityValidationError(f"{context}: 'conditions' must be a list.")
171:     conditions: list[Condition] = []
