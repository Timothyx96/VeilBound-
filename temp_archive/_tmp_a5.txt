302: def resolve_character_reference(definitions: Mapping[str, Any], reference: str) -> str | None:
303:     """Resolve a JSON character reference to a roster id (id first, then exact name)."""
304:     if reference in definitions:
305:         return reference
306:     for card_id, definition in definitions.items():
307:         if getattr(definition, "name", None) == reference:
308:             return card_id
309:     return None

312: def validate_ability_references(definitions: Mapping[str, Any]) -> None:
313:     """Validate every loaded ability against the full roster; raises on problems."""
314:     seen_ids: dict[str, str] = {}
315:     for card_id, definition in definitions.items():
316:         ability = getattr(definition, "ability", None)
317:         if ability is None:
318:             continue
319:         if ability.id in seen_ids:
320:             raise AbilityValidationError(
321:                 f"Duplicate ability id {ability.id!r} used by characters "
322:                 f"{seen_ids[ability.id]!r} and {card_id!r}."
323:             )
324:         seen_ids[ability.id] = card_id
325:         for condition in ability.conditions:
326:             if resolve_character_reference(definitions, condition.character) is None:
327:                 raise AbilityValidationError(
328:                     f"Ability {ability.id!r} on character {card_id!r} references unknown "
329:                     f"character {condition.character!r}; use a roster id or an exact name."
330:                 )
