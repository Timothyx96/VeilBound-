49: class CardInstance(StateRecord):
50:     """One dealt/played card.
51: 
52:     ``base_points`` lives on the frozen definition and never changes;
53:     ``current_points`` is this instance's runtime value, initialized from the
54:     definition when the match deals the card and only ever modified by
55:     ability effects. ``activated_abilities`` records which abilities this
56:     specific instance has already used during the match.
57:     """
58: 
59:     def __init__(
60:         self,
61:         id: str,
62:         definition_id: str,
63:         owner_id: str,
64:         revealed: bool = False,
65:         current_points: int | None = None,
66:         activated_abilities: list[str] | None = None,
67:         eliminated: bool = False,
68:         base_points: int | None = None,
69:         applied_bonuses: dict[str, int] | None = None,
70:     ):
71:         super().__init__(
72:             id=id,
73:             definition_id=definition_id,
74:             owner_id=owner_id,
75:             revealed=revealed,
76:             current_points=current_points,
77:             activated_abilities=list(activated_abilities or []),
78:             eliminated=eliminated,
79:             base_points=base_points,
80:             # Dynamic bonuses granted by abilities (ability_id -> amount). Kept
81:             # separate from the immutable base so removing a bonus never
82:             # corrupts the character's original points.
83:             applied_bonuses=dict(applied_bonuses) if applied_bonuses is not None else {},
84:         )

171:     @property
