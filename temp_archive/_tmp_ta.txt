89: def make_engine(capacity=2, seed=4):
90:     engine = GameEngine(
91:         GameConfig(
92:             deck_size=40,
93:             initial_hand_size=5,
94:             draw_per_turn=1,
95:             capacity_min=capacity,
96:             capacity_max=capacity,
97:             rounds_per_match=2,
98:         ),
99:         seed=seed,
100:     )
101:     # Scrub random pools so only scripted cards can carry duo synergies.
102:     # The roster now has several Dynamic Duo abilities; random draw piles
103:     # would otherwise leak unrelated activations into exact-count asserts.
104:     filler_ids = NON_ABILITY_FILLERS
105:     serial = 0
106: 
107:     def _filler(owner_id: str) -> CardInstance:
108:         nonlocal serial
109:         definition_id = filler_ids[serial % len(filler_ids)]
110:         serial += 1
111:         return instance(engine, definition_id, owner_id)
112: 
113:     for player in engine.state.players:
114:         player.draw_pile = [_filler(player.id) for _ in player.draw_pile]
115:         player.hand = [_filler(player.id) for _ in player.hand]
116:     return engine

119: def pick_filler(player):
120:     card = next((c for c in player.hand if c.definition_id not in SPECIAL), None)
121:     assert card is not None, "scenario needs a non-duo filler card in hand"
122:     return card

125: def place(engine, card, row_index, slot_index):
126:     active = engine.state.active_player
127:     assert active.id == card.owner_id, "test sequence must follow turn order"
128:     result = engine.play_card(active.id, card.id, engine.state.rows[row_index].id, slot_index)
129:     assert result.accepted, result.reason
130:     return result
131: 
132: 
133: def fill_rest(engine):
134:     """Finish the placement phase with filler cards (turns alternate)."""
135:     results = []
136:     while engine.state.phase == GamePhase.PLACEMENT:
137:         resolve_eliminations(engine)
138:         if engine.state.phase != GamePhase.PLACEMENT:
139:             break  # resolving an elimination closed the placement phase
140:         active = engine.state.active_player
141:         card = pick_filler(active)
142:         row_id, slot_index = engine.legal_slots()[0]
143:         result = engine.play_card(active.id, card.id, row_id, slot_index)
144:         assert result.accepted, result.reason
145:         results.append(result)
146:     resolve_eliminations(engine)
147:     return results
148: 
149: 
150: def finish_reveals(engine):
151:     """Run the final reveal sequence; returns one result per revealed card."""
152:     results = []
153:     while engine.state.phase == GamePhase.REVEAL:
154:         resolve_eliminations(engine)
155:         if engine.state.phase != GamePhase.REVEAL:
156:             # Resolving the final reveal's elimination completed the pass.
157:             break
158:         result = engine.reveal_next_card()
159:         assert result.accepted, result.reason
160:         results.append(result)
161:     resolve_eliminations(engine)
162:     assert engine.state.phase == GamePhase.ROW_SCORING
163:     return results
