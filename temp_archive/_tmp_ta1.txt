1|     # Scrub random pools so only scripted cards can carry duo synergies.
2|     # The roster now has several Dynamic Duo abilities; random draw piles
3|     # would otherwise leak unrelated activations into exact-count asserts.
4|     filler_ids = NON_ABILITY_FILLERS
5|     serial = 0
6| 
7|     def _filler(owner_id: str) -> CardInstance:
8|         nonlocal serial
9|         definition_id = filler_ids[serial % len(filler_ids)]
10|         serial += 1
11|         return instance(engine, definition_id, owner_id)
12| 
13|     for player in engine.state.players:
14|         player.draw_pile = [_filler(player.id) for _ in player.draw_pile]
15|         player.hand = [_filler(player.id) for _ in player.hand]
16|     return engine
17| 
18| 
19| def pick_filler(player):
20|     card = next((c for c in player.hand if c.definition_id not in SPECIAL), None)
21|     assert card is not None, "scenario needs a non-duo filler card in hand"
22|     return card

25| def place(engine, card, row_index, slot_index):
26|     active = engine.state.active_player
27|     assert active.id == card.owner_id, "test sequence must follow turn order"
28|     result = engine.play_card(active.id, card.id, engine.state.rows[row_index].id, slot_index)
29|     assert result.accepted, result.reason
30|     return result

33| def fill_rest(engine):
34|     """Finish the placement phase with filler cards (turns alternate)."""
35|     results = []
36|     while engine.state.phase == GamePhase.PLACEMENT:
37|         resolve_eliminations(engine)
38|         if engine.state.phase != GamePhase.PLACEMENT:
39|             break  # resolving an elimination closed the placement phase
40|         active = engine.state.active_player
41|         card = pick_filler(active)
42|         row_id, slot_index = engine.legal_slots()[0]
43|         result = engine.play_card(active.id, card.id, row_id, slot_index)
44|         assert result.accepted, result.reason
45|         results.append(result)
46|     resolve_eliminations(engine)
47|     return results

50| def finish_reveals(engine):
51|     """Run the final reveal sequence; returns one result per revealed card."""
52|     results = []
53|     while engine.state.phase == GamePhase.REVEAL:
54|         resolve_eliminations(engine)
55|         if engine.state.phase != GamePhase.REVEAL:
56|             # Resolving the final reveal's elimination completed the pass.
57|             break
58|         result = engine.reveal_next_card()
59|         assert result.accepted, result.reason
60|         results.append(result)
61|     resolve_eliminations(engine)
62|     assert engine.state.phase == GamePhase.ROW_SCORING
63|     return results

71|         for result in results
72|         for event in result.events
73|         if event.kind == "ability_activated"
