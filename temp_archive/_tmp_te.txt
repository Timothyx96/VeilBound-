34: def scripted_engine(capacity=3, seed=4):
35:     """Engine with scrubbed pools so only scripted cards reach the board."""
36:     engine = GameEngine(
37:         GameConfig(
38:             deck_size=40, initial_hand_size=5, draw_per_turn=1,
39:             capacity_min=capacity, capacity_max=capacity, rounds_per_match=2,
40:         ),
41:         seed=seed,
42:     )
43:     serial = 0
44: 
45:     def _scrub(player):
46:         nonlocal serial
47:         for index in range(len(player.draw_pile)):
48:             serial += 1
49:             definition_id = FILLERS[index % len(FILLERS)]
50:             player.draw_pile[index] = CardInstance(
51:                 f"filler-{serial}", definition_id, player.id,
52:                 current_points=engine.cards[definition_id].base_points,
53:             )
54: 
55:     for player in engine.state.players:
56:         _scrub(player)
57:     return engine
58:

60: def give(player, engine, *definition_ids):
61:     player.hand = []
62:     for definition_id in definition_ids:
63:         definition = engine.cards[definition_id]
64:         player.hand.append(CardInstance(
65:             f"{player.id}-{definition_id}-{len(player.hand)}",
66:             definition_id, player.id, current_points=definition.base_points,
67:         ))

70: def place(engine, card, row_index, slot_index):
71:     active = engine.state.active_player
72:     assert active.id == card.owner_id, "test sequence must follow turn order"
73:     result = engine.play_card(active.id, card.id, engine.state.rows[row_index].id, slot_index)
74:     assert result.accepted, result.reason
75:     return result

78: def card(engine, definition_id, owner_id):
79:     definition = engine.cards[definition_id]
80:     return CardInstance(
81:         f"unit-{definition_id}-{owner_id}", definition_id, owner_id,
82:         current_points=definition.base_points,
83:     )

86: def reveal_all_but(engine):
87:     """Reveal every placed card so only scripted hidden states remain."""
88:     for row in engine.state.rows:
89:         for slot in row.slots:
90:             if slot is not None and not slot.revealed:
91:                 slot.revealed = True

94: def targets_of(manager, state, attacker, targeting):
95:     return [
96:         (row.id, index, placed.definition_id)
97:         for row, index, placed in manager.get_valid_targets(state, attacker, targeting)
98:     ]

103: def _board_with_two_sides():
104:     """Capacity-2 board: p1 holds row0/s0 + row1/s0, p2 holds row0/s1 + row1/s1."""
105:     engine = scripted_engine(capacity=2)
106:     p1, p2 = engine.state.players
107:     a = card(engine, KAKASHI, p1.id)    # 7 / Fighter, the attacker
108:     p1_filler = card(engine, ITACHI, p1.id)
109:     b = card(engine, GO, p2.id)         # 10 / GOAT
110:     c = card(engine, HISOKA, p2.id)     # 8 / Fighter
111:     p1.hand = [a, p1_filler]
112:     p2.hand = [b, c]
113:     place(engine, a, 0, 0)
114:     place(engine, b, 0, 1)          # reveals a
115:     place(engine, p1_filler, 1, 0)
116:     place(engine, c, 1, 1)          # reveals p1_filler
117:     reveal_all_but(engine)
118:     return engine, a, b, c

202:     while engine.state.phase == GamePhase.PLACEMENT:

209:         assert result.accepted, result.reason

225: def test_final_reveal_pause_resumes_after_selecting_a_target():
226:     engine = scripted_engine(capacity=3)
227:     p1, p2 = engine.state.players
228:     give(p1, engine, GO, LUFFY, KILLUA)
229:     give(p2, engine, KAKASHI, HISOKA, ITACHI)
230:     go, luffy, killua = p1.hand
231:     kakashi, hisoka, itachi = p2.hand
232:     place(engine, go, 0, 0)              # GO (p1, hidden)
233:     place(engine, kakashi, 0, 1)         # KAKASHI reveals GO
234:     place(engine, killua, 0, 2)          # KILLUA reveals KAKASHI; row0 full
235:     place(engine, hisoka, 1, 0)
236:     place(engine, luffy, 1, 1)           # reveals HISOKA
237:     place(engine, itachi, 1, 2)          # reveals LUFFY
238:     fill_rest(engine)
239:     assert engine.state.phase == GamePhase.REVEAL
240: 
241:     reveal = engine.reveal_next_card()       # first hidden: row0/s2 = KILLUA
242:     assert reveal.accepted
243:     pending = engine.state.pending_elimination
244:     assert pending is not None
245:     assert pending["attacker_card_id"] == killua.id
246:     assert pending["attacker_owner_id"] == "player-1"
247:     # Row-major order: KAKASHI (row0/s1) first, then HISOKA (row1/s0).
248:     assert [(t["row_id"], t["slot_index"]) for t in pending["valid_targets"]] == [
249:         (engine.state.rows[0].id, 1),
250:         (engine.state.rows[1].id, 0),
251:     ]
252:     # The reveal flow is paused: another reveal is rejected until resolved.
253:     paused = engine.reveal_next_card()
254:     assert not paused.accepted
255:     assert "elimination" in paused.reason
256: 
257:     target = pending["valid_targets"][0]
258:     result = engine.select_elimination_target(target["row_id"], target["slot_index"])
259:     assert result.accepted, result.reason
260:     assert engine.state.pending_elimination is None
261:     eliminated = engine.state.rows[0].slots[1]
262:     assert eliminated.definition_id == KAKASHI
263:     assert eliminated.eliminated is True
264:     assert eliminated.current_points == 0
265:     assert engine.state.rows[0].eliminated_slots == [1]
266:     elimination_events = [event for event in result.events if event.kind == "elimination"]
267:     assert len(elimination_events) == 1
268:     assert elimination_events[0].data["attacker_name"] == "Killua Zoldyck"
269:     assert elimination_events[0].data["target_name"] == "Kakashi Hatake"
270: 
271:     # Reveal flow resumes and finishes the round normally.
272:     while engine.state.phase == GamePhase.REVEAL:
273:         assert engine.reveal_next_card().accepted
274:     assert engine.state.phase == GamePhase.ROW_SCORING
